//==============================================================
//  Bozzetto Exporter  —  ZBrush 2022+
//--------------------------------------------------------------
//  Exports a sculpt's UNDO-HISTORY steps (or manual snapshots)
//  as a sequential OBJ mesh sequence for the Bozzetto timelapse
//  viewer:  https://github.com/vidarrapp/bozzetto
//
//  Output:  <folder>/<prefix>0001.obj, <prefix>0002.obj, ...
//           The folder AND base name come from the Save dialog;
//           the base name is the frame prefix. Tool:Export merges
//           all VISIBLE SubTools into each file.
//
//  Decimation to Bozzetto's triangle budget is a SEPARATE,
//  optional step. See scripts/decimate_sequence.py (PyMeshLab) or
//  src/bozzetto_decimate_dm.txt (in-ZBrush Decimation Master).
//
//  INSTALL: Load this .txt via the ZScript palette (that compiles
//  a .zsc beside it). For auto-load on startup, copy the .zsc into
//  Pixologic/ZBrush 2022/ZStartup/ZPlugs64/. UI appears as a
//  "Bozzetto Exporter" subpalette in ZPlugin.
//
//  Command paths/args below are confirmed against the ZScript Command
//  Reference 2022 and live testing -- undo-history capture is validated
//  end to end (detect -> scrub -> export -> frames progress correctly).
//  Remaining //VERIFY items: Tool:Geometry:SDiv (subdivision-level
//  export) and the ImportExport export-toggle paths.
//==============================================================

[IConfig, 2022]

//--------------------------------------------------------------
//  Globals
//--------------------------------------------------------------
[VarDef, bozzPrefixDefault, "sculpt_"]
[VarDef, bozzPick, ""]
[VarDef, bozzDir, ""]
[VarDef, bozzPre, ""]
[VarDef, bozzName, ""]
[VarDef, bozzSD, 0]
[VarDef, bozzIdx, 1]
[VarDef, bozzUndo, ""]
[VarDef, bozzFound, ""]
[VarDef, bozzMaxStep, 0]
[VarDef, bozzMinStep, 0]
[VarDef, bozzStride, 1]
[VarDef, bozzCap, 0]
[VarDef, bozzCount, 0]
[VarDef, bozzNeed, 0]
[VarDef, bozzWas, 0]
[VarDef, bozzStep, 0]
[VarDef, bozzIters, 0]
[VarDef, bozzI, 0]
[VarDef, bozzPadNum, 0]
[VarDef, bozzPadWidth, 4]
[VarDef, bozzPadStr, ""]
[VarDef, bozzNamePath, ""]
[VarDef, bozzFrame1, ""]
[VarDef, bozzAns, 1]
// slashify scratch
[VarDef, slashIn, ""]
[VarDef, slashOut, ""]
[VarDef, slashLen, 0]
[VarDef, slashI, 0]
[VarDef, slashCh, ""]
[VarDef, slashCode, 0]
// export-pref switch helper
[VarDef, bozzSwName, ""]
[VarDef, bozzSwVal, 0]
[VarDef, bozzP1, ""]
[VarDef, bozzP2, ""]

//--------------------------------------------------------------
//  Persistent state in memblocks (survive button presses).
//     Bozz_State[0] = next frame number
//     Bozz_OutDir   = output folder (forward slashes, trailing /)
//     Bozz_Prefix   = filename prefix (from the Save-dialog name)
//     Bozz_UndoPath = detected Undo-Counter interface path
//--------------------------------------------------------------
[RoutineDef, bozzInit,
	[If, [MemGetSize, Bozz_State],,
		[MVarDef, Bozz_State, 4, 0]
		[MVarSet, Bozz_State, 0, 1]
	]
	[If, [MemGetSize, Bozz_OutDir],, [MemCreate, Bozz_OutDir, 1024, 0]]
	[If, [MemGetSize, Bozz_Prefix],,
		[MemCreate, Bozz_Prefix, 256, 0]
		[MemWriteString, Bozz_Prefix, bozzPrefixDefault, 0]
	]
	[If, [MemGetSize, Bozz_UndoPath],, [MemCreate, Bozz_UndoPath, 256, 0]]
]

//--------------------------------------------------------------
//  Set one ImportExport "Export" switch to bozzSwVal, probing the
//  two likely interface paths (with/without the "Export" group).
//--------------------------------------------------------------
[RoutineDef, bozzSetSwitch,
	[VarSet, bozzP1, [StrMerge, "Preferences:ImportExport:Export:", bozzSwName]]
	[VarSet, bozzP2, [StrMerge, "Preferences:ImportExport:", bozzSwName]]
	[If, [IExists, bozzP1],
		[ISet, bozzP1, bozzSwVal]
	,
		[If, [IExists, bozzP2], [ISet, bozzP2, bozzSwVal]]
	]
]

//--------------------------------------------------------------
//  Apply Bozzetto-friendly OBJ export options before exporting:
//    Txr off -> no UVs -> no .mtl file written
//    Mrg off -> don't merge UV coords
//    Qud off / Tri on -> triangulated (Bozzetto/three.js want tris)
//  Normal-flip options (eFlipNormals, eFlipMapVert, eSwitchYZ ...)
//  are left as you set them in Preferences>ImportExport>Export.
//--------------------------------------------------------------
[RoutineDef, bozzSetExportPrefs,
	[VarSet, bozzSwName, "Txr"] [VarSet, bozzSwVal, 0] [RoutineCall, bozzSetSwitch]
	[VarSet, bozzSwName, "Mrg"] [VarSet, bozzSwVal, 0] [RoutineCall, bozzSetSwitch]
	[VarSet, bozzSwName, "Qud"] [VarSet, bozzSwVal, 0] [RoutineCall, bozzSetSwitch]
	[VarSet, bozzSwName, "Tri"] [VarSet, bozzSwVal, 1] [RoutineCall, bozzSetSwitch]
]

//--------------------------------------------------------------
//  Convert backslashes in slashIn to forward slashes -> slashOut.
//--------------------------------------------------------------
[RoutineDef, slashify,
	[VarSet, slashOut, ""]
	[VarSet, slashLen, [StrLength, slashIn]]
	[If, slashLen > 0,
		[Loop, slashLen,
			[VarSet, slashCh, [StrExtract, slashIn, slashI, slashI]]
			[VarSet, slashCode, [StrToAsc, slashCh]]
			[If, slashCode == 92,
				[VarSet, slashOut, [StrMerge, slashOut, "/"]]
			,
				[VarSet, slashOut, [StrMerge, slashOut, slashCh]]
			]
		, slashI]
	]
]

//--------------------------------------------------------------
//  Zero-pad bozzPadNum to bozzPadWidth digits -> bozzPadStr
//--------------------------------------------------------------
[RoutineDef, bozzPad,
	[VarSet, bozzPadStr, [StrMerge, "", bozzPadNum]]
	[VarSet, bozzNeed, bozzPadWidth - [StrLength, bozzPadStr]]
	[If, bozzNeed > 0,
		[Loop, bozzNeed, [VarSet, bozzPadStr, [StrMerge, "0", bozzPadStr]] ]
	]
]

//--------------------------------------------------------------
//  Build the next frame's full path -> bozzNamePath
//--------------------------------------------------------------
[RoutineDef, bozzFrameName,
	[VarSet, bozzPadNum, [MVarGet, Bozz_State, 0]]
	[VarSet, bozzPadWidth, [IGet, "ZPlugin:Bozzetto Exporter:Filename Padding"]]
	[RoutineCall, bozzPad]
	[MemReadString, Bozz_OutDir, bozzDir, 0]
	[MemReadString, Bozz_Prefix, bozzPre, 0]
	[VarSet, bozzNamePath, [StrMerge, bozzDir, bozzPre, bozzPadStr, ".obj"]]
]

//--------------------------------------------------------------
//  Export the current visible mesh state as the next frame.
//--------------------------------------------------------------
[RoutineDef, bozzExportCurrent,
	[MemReadString, Bozz_OutDir, bozzDir, 0]
	[If, [StrLength, bozzDir] == 0,
		[Note, "Set the output folder first (Set Output Folder).", , 3]
	,
		[VarSet, bozzSD, [IGet, "ZPlugin:Bozzetto Exporter:Export Subdiv Level"]]
		[If, bozzSD > 0,
			[If, [IExists, "Tool:Geometry:SDiv"],
				[ISet, "Tool:Geometry:SDiv", bozzSD]   //VERIFY: path + active-level export
			]
		]
		[RoutineCall, bozzFrameName]
		[FileNameSetNext, bozzNamePath]
		[IPress, "Tool:Export"]
		[VarSet, bozzIdx, [MVarGet, Bozz_State, 0] + 1]
		[MVarSet, Bozz_State, 0, bozzIdx]
		[ISet, "ZPlugin:Bozzetto Exporter:Next Frame #", bozzIdx]
	]
]

//--------------------------------------------------------------
//  Probe for the per-SubTool Undo Counter control (Edit > Tool).
//--------------------------------------------------------------
[RoutineDef, bozzDetectUndo,
	[MemWriteString, Bozz_UndoPath, "", 0]
	[VarSet, bozzFound, ""]
	[If, [IExists, "Edit:Tool:UndoHistory"],   [VarSet, bozzFound, "Edit:Tool:UndoHistory"]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo History"], [VarSet, bozzFound, "Edit:Tool:Undo History"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo Counter"], [VarSet, bozzFound, "Edit:Tool:Undo Counter"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:UndoCounter"],  [VarSet, bozzFound, "Edit:Tool:UndoCounter"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo"],         [VarSet, bozzFound, "Edit:Tool:Undo"]]]
	[If, [StrLength, bozzFound] > 0,
		[MemWriteString, Bozz_UndoPath, bozzFound, 0]
		[Note, [StrMerge, "Undo Counter found: ", bozzFound, "   (max step = ", [IGetMax, bozzFound], ")"], , 5]
	,
		[Note, "Undo Counter control not found by auto-probe. Use Capture Stage, or tell me your Edit>Tool slider name.", , 6]
	]
]

//--------------------------------------------------------------
//  Walk the undo history oldest -> newest. Resets to frame 1
//  (full timeline = fresh sequence) and asks before overwriting.
//--------------------------------------------------------------
[RoutineDef, bozzExportHistory,
	[MemReadString, Bozz_OutDir, bozzDir, 0]
	[If, [StrLength, bozzDir] == 0,
		[Note, "Set the output folder first (Set Output Folder).", , 3]
	,
		[RoutineCall, bozzSetExportPrefs]
		// overwrite check: does frame 0001 already exist?
		[MemReadString, Bozz_Prefix, bozzPre, 0]
		[VarSet, bozzPadNum, 1]
		[VarSet, bozzPadWidth, [IGet, "ZPlugin:Bozzetto Exporter:Filename Padding"]]
		[RoutineCall, bozzPad]
		[VarSet, bozzFrame1, [StrMerge, bozzDir, bozzPre, bozzPadStr, ".obj"]]
		[VarSet, bozzAns, 1]
		[If, [FileExists, bozzFrame1],
			[VarSet, bozzAns, [MessageOKCancel, [StrMerge, "Frames already exist (", bozzPre, bozzPadStr, " ...). Overwrite and start a fresh sequence?"], "Bozzetto Exporter"]]
		]
		[If, bozzAns == 0,
			[Note, "Export cancelled.", , 3]
		,
			// fresh full-timeline export starts at frame 1
			[MVarSet, Bozz_State, 0, 1]
			[ISet, "ZPlugin:Bozzetto Exporter:Next Frame #", 1]
			// make sure we have an undo-counter path
			[MemReadString, Bozz_UndoPath, bozzUndo, 0]
			[If, [StrLength, bozzUndo] == 0,
				[RoutineCall, bozzDetectUndo]
				[MemReadString, Bozz_UndoPath, bozzUndo, 0]
			]
			[If, [StrLength, bozzUndo] == 0,
				[Note, "No Undo Counter control found. Use Capture Stage instead.", , 5]
			,
				[VarSet, bozzMaxStep, [IGetMax, bozzUndo]]
				[VarSet, bozzMinStep, [IGetMin, bozzUndo]]
				[If, bozzMaxStep <= bozzMinStep,
					[Note, "No undo history on the active SubTool. Raise Preferences>Undo History>Max Undo History.", , 6]
				,
					[VarSet, bozzStride, [IGet, "ZPlugin:Bozzetto Exporter:Every Nth Step"]]
					[If, bozzStride < 1, [VarSet, bozzStride, 1]]
					[VarSet, bozzCap, [IGet, "ZPlugin:Bozzetto Exporter:Max Frames (0=all)"]]
					[VarSet, bozzCount, bozzMaxStep - bozzMinStep]
					[If, bozzCap > 0,
						[VarSet, bozzNeed, (bozzCount / bozzCap) + 1]
						[If, bozzNeed > bozzStride, [VarSet, bozzStride, bozzNeed]]
					]
					[VarSet, bozzWas, [IGet, bozzUndo]]
					[VarSet, bozzIters, (bozzCount / bozzStride) + 1]
					[Loop, bozzIters,
						[VarSet, bozzStep, bozzMinStep + (bozzI * bozzStride)]
						[If, bozzStep <= bozzMaxStep,
							[ISet, bozzUndo, bozzStep]            // scrubs geometry to this step (validated end to end)
							[RoutineCall, bozzExportCurrent]
						]
					, bozzI]
					[ISet, bozzUndo, bozzWas]
					[Note, "Undo-history export complete.", , 4]
				]
			]
		]
	]
]

//--------------------------------------------------------------
//  Initialise state, then build the UI.
//--------------------------------------------------------------
[RoutineCall, bozzInit]

[ISubPalette, "ZPlugin:Bozzetto Exporter"]

[IButton, "ZPlugin:Bozzetto Exporter:Set Output Folder",
	"Browse to your output folder, set a base filename, and click Save. No file is created. Both the folder and the name are used: the name becomes the frame prefix (e.g. sculpt_ -> sculpt_0001.obj; dragon_ -> dragon_0001.obj).",
	// FileNameAsk is a SAVE dialog when a default name is given. We keep the
	// folder (FileNameExtract flag 1) AND the typed name as prefix (flag 2).
	[VarSet, bozzPick, [FileNameAsk, "*.obj", "sculpt_.obj", "Set OUTPUT FOLDER + base name, then Save (no file is created; the name becomes the frame prefix)"]]
	[If, [StrLength, bozzPick] > 0,
		[VarSet, bozzDir, [FileNameExtract, bozzPick, 1]]
		[VarSet, slashIn, bozzDir]
		[RoutineCall, slashify]
		[VarSet, bozzDir, slashOut]
		[VarSet, slashLen, [StrLength, bozzDir]]
		[If, slashLen > 0,
			[VarSet, slashCh, [StrExtract, bozzDir, slashLen - 1, slashLen - 1]]
			[If, [StrToAsc, slashCh] != 47, [VarSet, bozzDir, [StrMerge, bozzDir, "/"]]]
		]
		[MemWriteString, Bozz_OutDir, bozzDir, 0]
		[VarSet, bozzName, [FileNameExtract, bozzPick, 2]]
		[If, [StrLength, bozzName] == 0, [VarSet, bozzName, "sculpt_"]]
		[MemWriteString, Bozz_Prefix, bozzName, 0]
		[Note, [StrMerge, "Output: ", bozzDir, "    prefix: ", bozzName], , 5]
	]
]

[ISlider, "ZPlugin:Bozzetto Exporter:Export Subdiv Level", 0, 1, 0, 8,
	"Subdivision level to export (0 = keep the current level). Lower levels = smaller files."]

[ISlider, "ZPlugin:Bozzetto Exporter:Filename Padding", 4, 1, 1, 8,
	"Zero-padding width for the frame number (4 -> 0001)."]

[ISlider, "ZPlugin:Bozzetto Exporter:Every Nth Step", 1, 1, 1, 50,
	"Export every Nth undo step (1 = every step)."]

[ISlider, "ZPlugin:Bozzetto Exporter:Max Frames (0=all)", 0, 1, 0, 2000,
	"Cap the total frame count; the step stride is auto-derived (0 = no cap)."]

[ISlider, "ZPlugin:Bozzetto Exporter:Next Frame #", 1, 1, 1, 9999,
	"Number used for the next exported frame. Edit to resume an existing sequence.",
	[MVarSet, Bozz_State, 0, [IGet, "ZPlugin:Bozzetto Exporter:Next Frame #"]]
]

[IButton, "ZPlugin:Bozzetto Exporter:Detect Undo History",
	"Probe ZBrush for the Undo Counter control and report how many history steps the active SubTool has. Run this first.",
	[RoutineCall, bozzDetectUndo]
]

[IButton, "ZPlugin:Bozzetto Exporter:Export Undo History",
	"Step through the active SubTool's undo history and export one OBJ per (sub-sampled) step. Starts a fresh sequence at frame 1 (asks before overwriting).",
	[RoutineCall, bozzExportHistory]
]

[IButton, "ZPlugin:Bozzetto Exporter:Capture Stage",
	"Manually export the current state as the next frame. Reliable fallback, independent of undo history.",
	[RoutineCall, bozzSetExportPrefs]
	[RoutineCall, bozzExportCurrent]
]

[IButton, "ZPlugin:Bozzetto Exporter:New Sequence (Reset)",
	"Reset the frame counter to 1 to start a fresh sequence.",
	[MVarSet, Bozz_State, 0, 1]
	[ISet, "ZPlugin:Bozzetto Exporter:Next Frame #", 1]
	[Note, "Frame counter reset to 1.", , 3]
]
