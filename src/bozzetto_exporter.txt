//==============================================================
//  Bozzetto Exporter  —  ZBrush 2022+
//--------------------------------------------------------------
//  Exports a sculpt's UNDO-HISTORY steps (or manual snapshots)
//  as a sequential OBJ mesh sequence for the Bozzetto timelapse
//  viewer:  https://github.com/vidarrapp/bozzetto
//
//  Output:  <folder>/sculpt_0001.obj, sculpt_0002.obj, ...
//           One OBJ per frame; Tool:Export merges all VISIBLE
//           SubTools into that single file.
//
//  Decimation to Bozzetto's triangle budget is a SEPARATE,
//  optional step (it is not always needed — exporting at a lower
//  subdivision level is often enough). See:
//     scripts/decimate_sequence.py      (PyMeshLab, command line)
//     src/bozzetto_decimate_dm.txt      (in-ZBrush Decimation Master)
//
//  INSTALL
//    * Compile this .txt to .zsc (ZScript palette > Load this file,
//      then Save As .zsc) and drop the .zsc into
//        Pixologic/ZBrush 2022/ZStartup/ZPlugs64/
//      so it auto-loads, OR
//    * Preferences > ZPlugin/ZScript and Load the .zsc each session.
//    The UI appears as a "Bozzetto Exporter" subpalette in ZPlugin.
//
//  -------------------------------------------------------------
//  NOTES ON THINGS LEARNED THE HARD WAY
//  -------------------------------------------------------------
//  * MemReadString WRITES into a target variable; it does not
//    return a value:  [MemReadString, block, targetVar, offset]
//  * StrMerge / Note drop backslashes, and ZBrush accepts forward
//    slashes in paths, so we convert "\" -> "/" on input and use
//    forward slashes everywhere (see the slashify routine).
//  * Global vars can be flushed when another script loads, so
//    persistent state lives in memblocks.
//
//  Lines marked //VERIFY use command paths/args that still want a
//  sanity check on real hardware. The biggest unknown is the Undo
//  Counter interface path; the "Detect Undo History" button probes
//  for it — run that first.
//==============================================================

[IConfig, 2022]   //VERIFY: declares the UI was built for ZBrush 2022 sizing

//--------------------------------------------------------------
//  Global variables (shared by the routines below)
//--------------------------------------------------------------
[VarDef, bozzPrefixDefault, "sculpt_"]
[VarDef, bozzPick, ""]
[VarDef, bozzDir, ""]
[VarDef, bozzPre, ""]
[VarDef, bozzSD, 0]
[VarDef, bozzIdx, 1]
[VarDef, bozzUndo, ""]
[VarDef, bozzFound, ""]
[VarDef, bozzMaxStep, 0]
[VarDef, bozzMinStep, 0]
[VarDef, bozzStride, 1]
[VarDef, bozzCap, 0]
[VarDef, bozzCount, 0]
[VarDef, bozzNeed, 0]
[VarDef, bozzWas, 0]
[VarDef, bozzStep, 0]
[VarDef, bozzIters, 0]
[VarDef, bozzI, 0]
[VarDef, bozzPadNum, 0]
[VarDef, bozzPadWidth, 4]
[VarDef, bozzPadStr, ""]
[VarDef, bozzNamePath, ""]
// slashify scratch vars
[VarDef, slashIn, ""]
[VarDef, slashOut, ""]
[VarDef, slashLen, 0]
[VarDef, slashI, 0]
[VarDef, slashCh, ""]
[VarDef, slashCode, 0]

//--------------------------------------------------------------
//  Persistent state in memblocks (survive button presses).
//     Bozz_State[0] = next frame number (int)
//     Bozz_OutDir   = output directory string (forward slashes,
//                     incl. trailing "/")
//     Bozz_Prefix   = filename prefix string
//     Bozz_UndoPath = detected Undo-Counter interface path string
//--------------------------------------------------------------
[RoutineDef, bozzInit,
	[If, [MemGetSize, Bozz_State],,
		[MVarDef, Bozz_State, 4, 0]
		[MVarSet, Bozz_State, 0, 1]
	]
	[If, [MemGetSize, Bozz_OutDir],, [MemCreate, Bozz_OutDir, 1024, 0]]
	[If, [MemGetSize, Bozz_Prefix],,
		[MemCreate, Bozz_Prefix, 256, 0]
		[MemWriteString, Bozz_Prefix, bozzPrefixDefault, 0]
	]
	[If, [MemGetSize, Bozz_UndoPath],, [MemCreate, Bozz_UndoPath, 256, 0]]
]

//--------------------------------------------------------------
//  Convert backslashes in slashIn to forward slashes -> slashOut.
//  ZBrush accepts forward slashes in paths, and this avoids the
//  StrMerge/Note backslash-stripping problem.
//--------------------------------------------------------------
[RoutineDef, slashify,
	[VarSet, slashOut, ""]
	[VarSet, slashLen, [StrLength, slashIn]]
	[If, slashLen > 0,
		[Loop, slashLen,
			[VarSet, slashCh, [StrExtract, slashIn, slashI, slashI]]  //VERIFY: 0-based, inclusive single char
			[VarSet, slashCode, [StrToAsc, slashCh]]                  //VERIFY: ASCII code of the char (92 = "\")
			[If, slashCode == 92,
				[VarSet, slashOut, [StrMerge, slashOut, "/"]]
			,
				[VarSet, slashOut, [StrMerge, slashOut, slashCh]]
			]
		, slashI]
	]
]

//--------------------------------------------------------------
//  Zero-pad bozzPadNum to bozzPadWidth digits -> bozzPadStr
//--------------------------------------------------------------
[RoutineDef, bozzPad,
	[VarSet, bozzPadStr, [StrMerge, "", bozzPadNum]]   //VERIFY: numeric -> string coercion
	[VarSet, bozzNeed, bozzPadWidth - [StrLength, bozzPadStr]]
	[If, bozzNeed > 0,
		[Loop, bozzNeed,
			[VarSet, bozzPadStr, [StrMerge, "0", bozzPadStr]]
		]
	]
]

//--------------------------------------------------------------
//  Build the next frame's full path -> bozzNamePath
//     <dir><prefix><zero-padded index>.obj   (forward slashes)
//--------------------------------------------------------------
[RoutineDef, bozzFrameName,
	[VarSet, bozzPadNum, [MVarGet, Bozz_State, 0]]
	[VarSet, bozzPadWidth, [IGet, "ZPlugin:Bozzetto Exporter:Filename Padding"]]
	[RoutineCall, bozzPad]
	[MemReadString, Bozz_OutDir, bozzDir, 0]   // reads into bozzDir
	[MemReadString, Bozz_Prefix, bozzPre, 0]   // reads into bozzPre
	[VarSet, bozzNamePath, [StrMerge, bozzDir, bozzPre, bozzPadStr, ".obj"]]
]

//--------------------------------------------------------------
//  Export the current visible mesh state as the next frame.
//  Used by both "Capture Stage" and the undo-history loop.
//--------------------------------------------------------------
[RoutineDef, bozzExportCurrent,
	[MemReadString, Bozz_OutDir, bozzDir, 0]
	[If, [StrLength, bozzDir] == 0,
		[Note, "Set the output folder first (Set Output Folder).", , 3]   //VERIFY: Note args (text, handle, seconds)
	,
		// optional: set the subdivision level to export (0 = leave as-is)
		[VarSet, bozzSD, [IGet, "ZPlugin:Bozzetto Exporter:Export Subdiv Level"]]
		[If, bozzSD > 0,
			[If, [IExists, "Tool:Geometry:SDiv"],
				[ISet, "Tool:Geometry:SDiv", bozzSD]   //VERIFY: path + that export uses active level
			]
		]
		// name + export (Tool:Export merges all visible SubTools)
		[RoutineCall, bozzFrameName]
		[FileNameSetNext, bozzNamePath]
		[IPress, "Tool:Export"]                        // confirmed pattern
		// advance the counter and reflect it in the UI
		[VarSet, bozzIdx, [MVarGet, Bozz_State, 0] + 1]
		[MVarSet, Bozz_State, 0, bozzIdx]
		[ISet, "ZPlugin:Bozzetto Exporter:Next Frame #", bozzIdx]
	]
]

//--------------------------------------------------------------
//  Probe for the per-SubTool Undo Counter control. It lives in
//  Edit > Tool, but the exact ZScript token is version-specific,
//  so we try several and keep the first that exists.
//--------------------------------------------------------------
[RoutineDef, bozzDetectUndo,
	[MemWriteString, Bozz_UndoPath, "", 0]
	[VarSet, bozzFound, ""]
	[If, [IExists, "Edit:Tool:UndoHistory"],   [VarSet, bozzFound, "Edit:Tool:UndoHistory"]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo History"], [VarSet, bozzFound, "Edit:Tool:Undo History"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo Counter"], [VarSet, bozzFound, "Edit:Tool:Undo Counter"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:UndoCounter"],  [VarSet, bozzFound, "Edit:Tool:UndoCounter"]]]
	[If, [StrLength, bozzFound] == 0, [If, [IExists, "Edit:Tool:Undo"],         [VarSet, bozzFound, "Edit:Tool:Undo"]]]
	[If, [StrLength, bozzFound] > 0,
		[MemWriteString, Bozz_UndoPath, bozzFound, 0]
		[Note, [StrMerge, "Undo Counter found: ", bozzFound, "   (max step = ", [IGetMax, bozzFound], ")"], , 5]
	,
		[Note, "Undo Counter control not found by auto-probe. Note your Edit>Tool slider's exact name and add it to bozzDetectUndo, or use Capture Stage.", , 6]
	]
]

//--------------------------------------------------------------
//  Walk the undo history oldest -> newest, exporting one frame
//  per (sub-sampled) step, then restore the latest step.
//
//  NOTE: Tool:Export is synchronous, so a straight loop works.
//  IF on your hardware scrubbing the Undo Counter breaks script
//  flow (the way Decimation Master does), switch to the memblock
//  "replay" pattern used in src/bozzetto_decimate_dm.txt.
//--------------------------------------------------------------
[RoutineDef, bozzExportHistory,
	[MemReadString, Bozz_OutDir, bozzDir, 0]
	[If, [StrLength, bozzDir] == 0,
		[Note, "Set the output folder first (Set Output Folder).", , 3]
	,
		// make sure we have an undo-counter path (detect if needed)
		[MemReadString, Bozz_UndoPath, bozzUndo, 0]
		[If, [StrLength, bozzUndo] == 0,
			[RoutineCall, bozzDetectUndo]
			[MemReadString, Bozz_UndoPath, bozzUndo, 0]
		]
		[If, [StrLength, bozzUndo] == 0,
			[Note, "No Undo Counter control found. Use Capture Stage instead.", , 5]
		,
			[VarSet, bozzMaxStep, [IGetMax, bozzUndo]]
			[VarSet, bozzMinStep, [IGetMin, bozzUndo]]   //VERIFY: usually 0
			[If, bozzMaxStep <= bozzMinStep,
				[Note, "No undo history on the active SubTool. Raise Preferences>Undo History>Max Undo History and sculpt with history enabled.", , 6]
			,
				// stride from "Every Nth Step", widened to honour "Max Frames"
				[VarSet, bozzStride, [IGet, "ZPlugin:Bozzetto Exporter:Every Nth Step"]]
				[If, bozzStride < 1, [VarSet, bozzStride, 1]]
				[VarSet, bozzCap, [IGet, "ZPlugin:Bozzetto Exporter:Max Frames (0=all)"]]
				[VarSet, bozzCount, bozzMaxStep - bozzMinStep]
				[If, bozzCap > 0,
					[VarSet, bozzNeed, (bozzCount / bozzCap) + 1]
					[If, bozzNeed > bozzStride, [VarSet, bozzStride, bozzNeed]]
				]
				// remember current step to restore later
				[VarSet, bozzWas, [IGet, bozzUndo]]
				// iterate (extra +1 iteration is guarded by the inner If)
				[VarSet, bozzIters, (bozzCount / bozzStride) + 1]
				[Loop, bozzIters,
					[VarSet, bozzStep, bozzMinStep + (bozzI * bozzStride)]
					[If, bozzStep <= bozzMaxStep,
						[ISet, bozzUndo, bozzStep]            //VERIFY: scrubs geometry to this step
						[RoutineCall, bozzExportCurrent]
					]
				, bozzI]
				// restore the latest state
				[ISet, bozzUndo, bozzWas]
				[Note, "Undo-history export complete.", , 4]
			]
		]
	]
]

//--------------------------------------------------------------
//  Initialise state, then build the UI.
//--------------------------------------------------------------
[RoutineCall, bozzInit]

[ISubPalette, "ZPlugin:Bozzetto Exporter"]

[IButton, "ZPlugin:Bozzetto Exporter:Set Output Folder",
	"Choose the folder OBJ frames are written to. Browse INTO the folder, leave/type any name in the File name box, and click Open. No file is created -- only the folder is used. (You can also just select any existing file in that folder.)",
	// FileNameAsk doubles as a save dialog: the typed name need NOT exist,
	// so a prefilled default lets the user just click Open. We use only the
	// folder part (FileNameExtract flag 3).
	[VarSet, bozzPick, [FileNameAsk, "*.obj", "frames.obj", "Browse INTO your output folder, then click Open. The name need not exist; only the folder is used."]]
	[If, [StrLength, bozzPick] > 0,
		[VarSet, bozzDir, [FileNameExtract, bozzPick, 3]]   //VERIFY: flag 3 = drive+path (incl. trailing slash)
		[VarSet, slashIn, bozzDir]
		[RoutineCall, slashify]
		[VarSet, bozzDir, slashOut]
		[MemWriteString, Bozz_OutDir, bozzDir, 0]
		[Note, [StrMerge, "Output folder set: ", bozzDir], , 4]
	]
]

	"Reset the frame counter to 1 to start a fresh sequence.",

	[ISet, "ZPlugin:Bozzetto Exporter:Next Frame #", 1]
	[Note, "Frame counter reset to 1.", , 3]
]
